cmake_minimum_required(VERSION 3.21)
project(PmxVertexExportTests LANGUAGES CXX)
include(CTest)
get_filename_component(MMS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(MMV_PUBLIC_INCLUDE_DIR "${MMS_ROOT}/../MMV/include" CACHE PATH "MMV public headers")
if(NOT EXISTS "${MMV_PUBLIC_INCLUDE_DIR}/mmv/mmv.h")
    set(MMV_PUBLIC_INCLUDE_DIR "${MMS_ROOT}/include")
endif()
add_executable(pmx_vertex_export_tests ../pmx_vertex_export_tests.cpp)
target_include_directories(pmx_vertex_export_tests PRIVATE "${MMS_ROOT}/include" "${MMV_PUBLIC_INCLUDE_DIR}")
target_compile_features(pmx_vertex_export_tests PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(pmx_vertex_export_tests PRIVATE /utf-8)
endif()
add_test(NAME pmx_vertex_export COMMAND pmx_vertex_export_tests)
